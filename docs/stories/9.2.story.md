# Story 9.2: Product Detail Page

## Status

**TODO** - Ready for development

## Story

**As a** consumer,
**I want** to view a product's complete journey on a dedicated page,
**so that** I can verify its origin, see all supply chain events, and trust the product's authenticity.

## Acceptance Criteria

### Page Access
1. Page accessible at `/trace/[id]` route (public, no authentication)
2. Parameter `id` is the blockchainId (numeric)
3. No wallet connection prompt (no MetaMask, no RainbowKit)
4. Server-side rendering for SEO and fast initial load

### Product Header Section
5. Product name displayed prominently
6. Origin location displayed
7. Harvest date displayed (formatted for locale)
8. Producer company name displayed
9. Product blockchain ID displayed (e.g., "Product #42")

### Supply Chain Timeline
10. Display all trace records in chronological order (oldest first)
11. Each record shows: action badge, location, actor name, company name, timestamp
12. Each record includes Etherscan link for blockchain verification
13. Reuse existing TraceTimeline component from `src/components/trace/TraceTimeline.tsx`
14. Timeline fetches from existing `/api/products/[id]/trace-history` endpoint

### Navigation Actions
15. "Scan Another Product" button at bottom of page
16. Button navigates back to `/trace` landing page
17. No "Back" button (browser handles back navigation)

### Error States
18. Product not found (404) shows user-friendly message with link to `/trace`
19. Network error shows retry option
20. Loading state shows spinner while fetching data

### General
21. Use existing Layout component (no role prop)
22. Mobile-first responsive design (works on 360px width)
23. Touch targets minimum 44px

### Technical Specifications

**Page Component:**

```typescript
// src/pages/trace/[id].tsx
import { GetServerSidePropsContext } from 'next';
import { useRouter } from 'next/router';
import {
  Box,
  VStack,
  Heading,
  Text,
  Button,
  HStack,
  Badge,
  Divider,
  Center,
  Spinner,
  Alert,
  AlertIcon,
} from '@chakra-ui/react';
import { ViewIcon } from '@chakra-ui/icons';
import { Layout } from '@/components/layout';
import { TraceTimeline } from '@/components/trace';
import { prisma } from '@/lib/prisma';

interface Product {
  id: string;
  name: string;
  origin: string;
  blockchainId: number;
  harvestDate: string;
  transactionHash: string;
  company: { name: string };
}

interface ProductPageProps {
  product: Product | null;
  error?: string;
}

export async function getServerSideProps(context: GetServerSidePropsContext) {
  const { id } = context.params as { id: string };

  // Validate blockchainId is numeric
  const blockchainId = parseInt(id, 10);
  if (isNaN(blockchainId) || blockchainId <= 0) {
    return { props: { product: null, error: 'Invalid Product ID' } };
  }

  try {
    const product = await prisma.product.findFirst({
      where: { blockchainId },
      include: { company: { select: { name: true } } },
    });

    if (!product) {
      return { props: { product: null, error: 'Product not found' } };
    }

    return {
      props: {
        product: {
          id: product.id,
          name: product.name,
          origin: product.origin,
          blockchainId: product.blockchainId,
          harvestDate: product.harvestDate.toISOString(),
          transactionHash: product.transactionHash,
          company: product.company,
        },
      },
    };
  } catch (error) {
    console.error('Failed to fetch product:', error);
    return { props: { product: null, error: 'Failed to load product' } };
  }
}

export default function ProductTracePage({ product, error }: ProductPageProps) {
  const router = useRouter();

  // Error state
  if (error || !product) {
    return (
      <Layout>
        <Center py={16}>
          <VStack spacing={4}>
            <Alert status="warning" borderRadius="md">
              <AlertIcon />
              {error || 'Product not found'}
            </Alert>
            <Text color="brand.muted">
              The product you&apos;re looking for doesn&apos;t exist or has been removed.
            </Text>
            <Button colorScheme="green" onClick={() => router.push('/trace')}>
              Try Another Product
            </Button>
          </VStack>
        </Center>
      </Layout>
    );
  }

  return (
    <Layout>
      <VStack spacing={6} maxW="2xl" mx="auto" py={6} px={4}>
        {/* Product Header */}
        <Box w="full" bg="brand.surface" p={6} borderRadius="lg" borderWidth="1px">
          <VStack align="start" spacing={2}>
            <HStack justify="space-between" w="full">
              <Heading size="lg" color="brand.dark">
                {product.name}
              </Heading>
              <Badge colorScheme="green" fontSize="sm">
                Product #{product.blockchainId}
              </Badge>
            </HStack>
            <Text color="brand.muted">
              <strong>Origin:</strong> {product.origin}
            </Text>
            <Text color="brand.muted">
              <strong>Harvested:</strong>{' '}
              {new Date(product.harvestDate).toLocaleDateString()}
            </Text>
            <Text color="brand.muted">
              <strong>Producer:</strong> {product.company.name}
            </Text>
          </VStack>
        </Box>

        <Divider />

        {/* Journey Timeline */}
        <Box w="full">
          <Heading size="md" color="brand.dark" mb={4}>
            Product Journey
          </Heading>
          <TraceTimeline productId={product.id} />
        </Box>

        <Divider />

        {/* Actions */}
        <Button
          size="lg"
          colorScheme="green"
          variant="outline"
          leftIcon={<ViewIcon />}
          onClick={() => router.push('/trace')}
          w="full"
        >
          Scan Another Product
        </Button>
      </VStack>
    </Layout>
  );
}
```
