### Technical Specifications

**Theme Structure:**

```typescript
// src/styles/theme.ts
import { extendTheme } from '@chakra-ui/react';

const theme = extendTheme({
  colors: {
    brand: {
      primary: '#38A169',    // Green - main actions
      secondary: '#3182CE',  // Blue - links/secondary actions
      accent: '#DD6B20',     // Orange - warnings/highlights
      dark: '#1A202C',       // Dark gray - text
      light: '#F7FAFC',      // Light gray - backgrounds
    },
  },
  fonts: {
    heading: 'Inter, system-ui, sans-serif',
    body: 'Inter, system-ui, sans-serif',
  },
  breakpoints: {
    sm: '480px',
    md: '768px',
    lg: '1024px',
    xl: '1280px',
  },
  components: {
    Button: {
      defaultProps: {
        colorScheme: 'green',
      },
    },
    Input: {
      defaultProps: {
        focusBorderColor: 'green.500',
      },
    },
    Card: {
      baseStyle: {
        container: {
          borderRadius: 'md',
          boxShadow: 'sm',
        },
      },
    },
  },
});

export default theme;
```

**Layout Component:**

```typescript
// src/components/layout/Layout.tsx
import { Box, Flex } from '@chakra-ui/react';
import { ReactNode } from 'react';

interface LayoutProps {
  children: ReactNode;
}

export function Layout({ children }: LayoutProps) {
  return (
    <Flex direction="column" minH="100vh">
      <Box as="header" bg="brand.primary" color="white" p={4}>
        <Box maxW="xl" mx="auto">
          FoodTrace
        </Box>
      </Box>
      <Box as="main" flex="1" py={8}>
        <Box maxW="xl" mx="auto" px={{ base: 4, md: 8 }}>
          {children}
        </Box>
      </Box>
    </Flex>
  );
}
```

**LoadingSpinner Component:**

```typescript
// src/components/ui/LoadingSpinner.tsx
import { Center, Spinner, Text, VStack } from '@chakra-ui/react';

interface LoadingSpinnerProps {
  text?: string;
  size?: 'sm' | 'md' | 'lg' | 'xl';
}

export function LoadingSpinner({ text, size = 'lg' }: LoadingSpinnerProps) {
  return (
    <Center py={8}>
      <VStack spacing={4}>
        <Spinner size={size} color="brand.primary" thickness="4px" />
        {text && <Text color="gray.600">{text}</Text>}
      </VStack>
    </Center>
  );
}
```
