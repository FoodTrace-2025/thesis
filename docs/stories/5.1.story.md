### Technical Specifications

**Contract Structure (from Epic 5 PRD):**

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.20;

import "@openzeppelin/contracts/access/AccessControl.sol";

contract ProductRegistry is AccessControl {
    bytes32 public constant PRODUCER_ROLE = keccak256("PRODUCER_ROLE");

    struct Product {
        uint256 id;
        string name;
        string origin;
        uint256 harvestDate;
        address producer;
        uint256 timestamp;
        bool exists;
    }

    uint256 public productCount;
    mapping(uint256 => Product) public products;

    event ProductRegistered(
        uint256 indexed productId,
        address indexed producer,
        string name,
        uint256 timestamp
    );

    constructor() {
        _grantRole(DEFAULT_ADMIN_ROLE, msg.sender);
    }

    function registerProduct(
        string memory name,
        string memory origin,
        uint256 harvestDate
    ) public onlyRole(PRODUCER_ROLE) returns (uint256) {
        require(bytes(name).length > 0, "Name required");
        require(harvestDate <= block.timestamp, "Future date not allowed");

        productCount++;
        products[productCount] = Product({
            id: productCount,
            name: name,
            origin: origin,
            harvestDate: harvestDate,
            producer: msg.sender,
            timestamp: block.timestamp,
            exists: true
        });

        emit ProductRegistered(productCount, msg.sender, name, block.timestamp);
        return productCount;
    }

    function getProduct(uint256 productId) public view returns (Product memory) {
        require(products[productId].exists, "Product does not exist");
        return products[productId];
    }

    function productExists(uint256 productId) public view returns (bool) {
        return products[productId].exists;
    }

    function grantProducerRole(address account) public onlyRole(DEFAULT_ADMIN_ROLE) {
        _grantRole(PRODUCER_ROLE, account);
    }

    function revokeProducerRole(address account) public onlyRole(DEFAULT_ADMIN_ROLE) {
        _revokeRole(PRODUCER_ROLE, account);
    }

    function hasProducerRole(address account) public view returns (bool) {
        return hasRole(PRODUCER_ROLE, account);
    }
}
```

### Solidity Coding Standards (from architecture/coding-standards.md)

- **Version Pragma:** Use fixed version `pragma solidity 0.8.20;` (not floating ^0.8.0)
- **Contract Structure Order:** License → Pragma → Imports → Contract
- **Function Visibility Order:** Constructor → External → Public → Internal → Private
- **Security Patterns:** Use OpenZeppelin AccessControl, validate inputs with require
- **Gas Optimization:** Pack struct variables efficiently, use events for logs

### Test Structure Example

```typescript
import { expect } from "chai";
import { ethers } from "hardhat";
import { ProductRegistry } from "../typechain-types";

describe("ProductRegistry", function () {
  let productRegistry: ProductRegistry;
  let owner: any;
  let producer: any;
  let nonProducer: any;

  beforeEach(async function () {
    [owner, producer, nonProducer] = await ethers.getSigners();

    const ProductRegistry = await ethers.getContractFactory("ProductRegistry");
    productRegistry = await ProductRegistry.deploy();

    // Grant producer role
    await productRegistry.grantProducerRole(producer.address);
  });

  describe("registerProduct", function () {
    it("should register product with valid data", async function () {
      const harvestDate = Math.floor(Date.now() / 1000) - 86400; // Yesterday

      const tx = await productRegistry
        .connect(producer)
        .registerProduct("Organic Milk", "Oulu Farm", harvestDate);

      const receipt = await tx.wait();
      expect(receipt.status).to.equal(1);

      const product = await productRegistry.getProduct(1);
      expect(product.name).to.equal("Organic Milk");
      expect(product.origin).to.equal("Oulu Farm");
      expect(product.producer).to.equal(producer.address);
    });

    it("should emit ProductRegistered event", async function () {
      const harvestDate = Math.floor(Date.now() / 1000) - 86400;

      await expect(
        productRegistry
          .connect(producer)
          .registerProduct("Test Product", "Test Origin", harvestDate)
      )
        .to.emit(productRegistry, "ProductRegistered")
        .withArgs(1, producer.address, "Test Product", anyValue);
    });

    it("should reject empty product name", async function () {
      const harvestDate = Math.floor(Date.now() / 1000) - 86400;

      await expect(
        productRegistry.connect(producer).registerProduct("", "Oulu", harvestDate)
      ).to.be.revertedWith("Name required");
    });

    it("should reject future harvest date", async function () {
      const futureDate = Math.floor(Date.now() / 1000) + 86400; // Tomorrow

      await expect(
        productRegistry
          .connect(producer)
          .registerProduct("Milk", "Oulu", futureDate)
      ).to.be.revertedWith("Future date not allowed");
    });

    it("should reject non-producer", async function () {
      const harvestDate = Math.floor(Date.now() / 1000) - 86400;

      await expect(
        productRegistry
          .connect(nonProducer)
          .registerProduct("Milk", "Oulu", harvestDate)
      ).to.be.reverted;
    });
  });
});
```
