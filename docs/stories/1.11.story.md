### Technical Specifications

**Deployment Script (scripts/deploy.ts):**
```typescript
import { ethers } from "hardhat";

async function main() {
  const currentTimestampInSeconds = Math.round(Date.now() / 1000);
  const unlockTime = currentTimestampInSeconds + 60; // 1 minute from now

  const lockedAmount = ethers.parseEther("0.001");

  console.log(`Deploying Lock contract with unlock time: ${unlockTime}`);

  const Lock = await ethers.getContractFactory("Lock");
  const lock = await Lock.deploy(unlockTime, { value: lockedAmount });

  await lock.waitForDeployment();

  const address = await lock.getAddress();
  console.log(`Lock deployed to: ${address}`);
  console.log(`Unlock time: ${new Date(unlockTime * 1000).toISOString()}`);

  return { address, unlockTime };
}

main().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
```

**Verification Command:**
```bash
npx hardhat verify --network sepolia DEPLOYED_ADDRESS UNLOCK_TIME
```
